'' : 's'}.`); return count; } if (!data || !data.entries) throw new Error('invalid lorebook format: missing "entries"'); const entryCount = Object.keys(data.entries).length; await this.importSingleLorebookData(data, fallbackName); if (this.currentView === 'lore') this.renderLoreManagement(); alert(`Lorebook imported: ${fallbackName} (${entryCount} entries)`); return 1; } showLorebookImportOptions() { this.showBottomSheet([ { label: '📂 Import from File', action: () => document.getElementById('lorebookFileInput').click() }, { label: '🌐 Import from URL', action: () => this.openLorebookUrlModal() }, { label: '☁️📂 Import from Cloud', action: () => this.openCloudImportModal() } ]); } _normalizeLorebookUrl(url) { // Turn a GitHub Gist page URL into its raw content URL. const gist = url.match(/^https?:\/\/gist\.github\.com\/[^\/]+\/[a-f0-9]+\/?$/i); if (gist) return url.replace(/\/$/, '') + '/raw'; return url; } _lorebookNameFromUrl(url) { try { const parts = url.pathname.split('/').filter(Boolean); let last = parts.length ? decodeURIComponent(parts[parts.length - 1]) : ''; last = last.replace(/\.json$/i, ''); return last || url.hostname || 'Imported Lorebook'; } catch (e) { return 'Imported Lorebook'; } } openLorebookUrlModal() { let overlay = document.getElementById('loreUrlModal'); if (!overlay) { overlay = document.createElement('div');
